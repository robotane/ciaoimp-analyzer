    src.annotation.annotate.__main__ ¶

    Infer CasLog annotations for normalized Prolog contracts.

    This module produces build/annotated files by prepending CasLog mode/2 and measure/2 directives to normalized Prolog. It removes execution-only predicates such as goal/1 and main/0 because those are used by TNF execution scenarios, not by CasLog analysis.

    The main pipeline passes d-list rewrite hints in memory. No sidecar JSON file is required.

    def remove_execution_predicates(text: str) -> str:
    """Remove execution-only predicates before CasLog annotation.

    Step 6 is the CasLog input. It deliberately excludes:
    - goal/1, used as an execution scenario for TNF
    - main/0, when present as a local test harness
    """
    return "".join(
        clause for clause in _split_source_clauses(text)
        if not _is_goal_clause(clause) and not _is_main_clause(clause)
    ).lstrip()

    def apply_mode_overrides_to_file(path: Path) -> None:
    """Re-apply targeted overrides after mode demotion.

    ``demote_fixpoint`` may legitimately change helper modes, but for a few
    contract-specific helpers we want the final annotated file to keep a known
    stable mode shape.
    """
    file_overrides = MODE_OVERRIDES.get(path.stem)
    if not file_overrides:
        return
    text = path.read_text()

    def repl(match: re.Match) -> str:
        key = (match.group(2), int(match.group(3)))
        modes = file_overrides.get(key)
        if not modes:
            return match.group(0)
        return match.group(1) + ", ".join(modes) + match.group(5)

    updated = MODE_LINE_RE.sub(repl, text)
    if updated != text:
        path.write_text(updated)

    def annotate(src: Path, dst: Path, hints: dict | None = None) -> list[str]:
    """Annotate ``src`` into ``dst`` and return generated directive strings."""
    text = remove_execution_predicates(src.read_text())
    parsed = parse_clauses(text)
    forced_outputs, forced_lengths = _normalize_hints(hints)

    grouped: dict[tuple[str, int], list[tuple[str, list[str], str]]] = defaultdict(list)
    order: list[tuple[str, int]] = []
    seen: set[tuple[str, int]] = set()
    for nm, args, body in parsed:
        key = (nm, len(args))
        grouped[key].append((nm, args, body))
        if key not in seen:
            seen.add(key)
            order.append(key)

    known = set(grouped.keys())

    modes_map: dict[tuple[str, int], list[str]] = {}
    evidence_map: dict[tuple[str, int], list[set[str]]] = {}
    for key in order:
        name, arity = key
        preds = grouped[key]
        modes = infer_modes(name, arity, preds)
        for p in forced_outputs.get(key, ()):
            if 0 <= p < arity:
                modes[p] = "-"
        modes_map[key] = modes
        ev = _collect_evidence(name, arity, preds)
        ev = _share_by_variable(name, arity, preds, ev)
        evidence_map[key] = ev

    _apply_mode_overrides(src, modes_map)

    measures_map: dict[tuple[str, int], list[str]] = {
        key: _finalize_measures(key[0], evidence_map[key]) for key in order
    }

    # Mode backward propagation fixpoint
    for _ in range(20):
        changed = False
        for key in order:
            name, arity = key
            preds = grouped[key]
            for nm, args, body in preds:
                if nm != name:
                    continue
                var_pos: dict[str, list[int]] = defaultdict(list)
                for i, raw in enumerate(args):
                    if i >= arity:
                        break
                    a = raw.strip()
                    if VAR_RE.match(a) and a != "_":
                        var_pos[a].append(i)
                for cname, cargs in _find_body_calls(body, known):
                    cm = modes_map.get((cname, len(cargs)))
                    if cm is None:
                        continue
                    for i, raw in enumerate(cargs):
                        if i >= len(cm):
                            break
                        a = raw.strip()
                        if not (VAR_RE.match(a) and a != "_"):
                            continue
                        if cm[i] != "-":
                            continue
                        for hp in var_pos.get(a, []):
                            if modes_map[key][hp] != "-":
                                modes_map[key][hp] = "-"
                                changed = True
        if not changed:
            break

    # Seed forced length hints before measure fixpoint
    for key, positions in forced_lengths.items():
        if key not in evidence_map:
            continue
        for p in positions:
            if 0 <= p < len(evidence_map[key]):
                evidence_map[key][p].add("length")
    measures_map = {
        key: _finalize_measures(key[0], evidence_map[key]) for key in order
    }

    # Measure fixpoint: propagate callee measures to caller head positions
    for _ in range(20):
        changed = False
        for key in order:
            name, arity = key
            preds = grouped[key]
            for nm, args, body in preds:
                if nm != name:
                    continue
                var_pos: dict[str, list[int]] = defaultdict(list)
                for i, raw in enumerate(args):
                    if i >= arity:
                        break
                    for v in _vars_in(raw):
                        var_pos[v].append(i)
                for cname, cargs in _find_body_calls(body, known):
                    cm = measures_map.get((cname, len(cargs)))
                    if cm is None:
                        continue
                    for i, raw in enumerate(cargs):
                        if i >= len(cm):
                            break
                        a = raw.strip()
                        if not (VAR_RE.match(a) and a != "_"):
                            continue
                        positions = var_pos.get(a)
                        if not positions:
                            continue
                        m = cm[i]
                        if m == "void":
                            continue
                        for hp in positions:
                            if m not in evidence_map[key][hp]:
                                evidence_map[key][hp].add(m)
                                changed = True
            new_m = _finalize_measures(name, evidence_map[key])
            if new_m != measures_map[key]:
                measures_map[key] = new_m
                changed = True
        if not changed:
            break

    # Apply forced length hints after fixpoint (override _finalize_measures blanking)
    for key, positions in forced_lengths.items():
        if key not in measures_map:
            continue
        for p in positions:
            if 0 <= p < len(measures_map[key]):
                measures_map[key][p] = "length"

    directives: list[str] = []
    for key in order:
        name, arity = key
        modes = modes_map[key]
        measures = measures_map[key]
        directives.append(
            f":- mode({name}/{arity}, [{', '.join(modes)}]).\n"
            f":- measure({name}/{arity}, [{', '.join(measures)}]).\n"
        )

    out = "\n".join(directives) + "\n\n" + text
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text(out)
    return directives

    def main() -> int:
    """CLI wrapper for manual annotation/debugging."""

        _run_defaults()
        return 0

    def infer_modes(
    name: str, arity: int, clauses: list[tuple[str, list[str], str]]
) -> list[str]:
    """Infer mode list for predicate (name/arity)."""
    if arity == 0:
        return []
    if name == "neq" and arity == 2:
        return ["+", "+"]
    if _is_fact(clauses, name):
        per_pos_vals: list[list[str]] = [[] for _ in range(arity)]
        for nm, args, _ in clauses:
            if nm != name:
                continue
            for i in range(arity):
                if i < len(args):
                    per_pos_vals[i].append(args[i].strip())

        def has_compound(v: str) -> bool:
            return v != "_" and not VAR_RE.match(v) and not INT_RE.match(v)

        if any(any(has_compound(v) for v in vals) for vals in per_pos_vals):
            return ["+"] * (arity - 1) + ["-"]

        modes: list[str] = []
        for vals in per_pos_vals:
            is_output = any(v != "_" and (INT_RE.match(v) or not VAR_RE.match(v)) for v in vals)
            modes.append("-" if is_output else "+")
        return modes

    out_positions = _detect_output_positions(name, arity, clauses)
    if out_positions:
        modes_ = ["+"] * arity
        for i in out_positions:
            modes_[i] = "-"
        return modes_

    if name.startswith("aux_disj_"):
        literal_outputs = _detect_literal_output_positions(name, arity, clauses)
        if literal_outputs:
            modes_ = ["+"] * arity
            for i in literal_outputs:
                modes_[i] = "-"
            return modes_
        return ["+"] * arity

    if name.startswith("aux_cond_"):
        return ["+"] * arity

    preds_args = [(args, body) for nm, args, body in clauses if nm == name]
    last = arity - 1
    last_bound = True
    for args, body in preds_args:
        if last >= len(args):
            last_bound = False
            break
        a = args[last].strip()
        if a == "_":
            last_bound = False
            break
        if VAR_RE.match(a):
            shared = any(
                j != last and j < arity and args[j].strip() == a
                for j in range(len(args))
            )
            if shared:
                continue
            if re.search(rf"\b{re.escape(a)}\b", body):
                continue
            last_bound = False
            break
    if last_bound:
        return ["+"] * (arity - 1) + ["-"]
    return ["+"] * arity

    def parse_body_goals(body: str) -> list[tuple[str, list[str]]]:
    """Parse a clause body into an ordered list of (name, args) goals."""
    out: list[tuple[str, list[str]]] = []
    if not body.strip():
        return out
    for g in split_top_level(body, ","):
        g = g.strip()
        if not g:
            continue
        if g == "!":
            out.append(("!", []))
            continue
        parsed = _parse_infix_goal(g)
        if parsed:
            out.append(parsed)
            continue
        m = re.match(r"^([a-z_][A-Za-z0-9_]*)\s*\((.*)\)\s*$", g, re.DOTALL)
        if m:
            out.append((m.group(1), tokenize_args(m.group(2))))
            continue
        m2 = re.match(r"^([a-z_][A-Za-z0-9_]*)\s*$", g)
        if m2:
            out.append((m2.group(1), []))
    return out

    def forward_dataflow_step(
    grouped: dict[tuple[str, int], list[tuple[str, list[str], str]]],
    order: list[tuple[str, int]],
    modes_map: dict[tuple[str, int], list[str]],
) -> bool:
    """One L→R pass over every clause body. Returns True on any change."""
    changed = False
    for key in order:
        name, arity = key
        for nm, args, body in grouped[key]:
            if nm != name:
                continue
            bound: set[str] = set()
            for i, raw in enumerate(args):
                if i >= arity:
                    break
                if modes_map[key][i] == "+":
                    bound |= _vars_in(raw)
            for gname, gargs in parse_body_goals(body):
                gkey = (gname, len(gargs))
                builtin = BUILTIN_MODES.get(gkey)
                gmodes = builtin if builtin is not None else modes_map.get(gkey)
                if gmodes is None:
                    for a in gargs:
                        bound |= _vars_in(a)
                    continue
                for i, raw in enumerate(gargs):
                    if i >= len(gmodes) or gmodes[i] != "+":
                        continue
                    vs = _vars_in(raw)
                    if not vs or vs.issubset(bound):
                        continue
                    if builtin is not None:
                        continue
                    if modes_map[gkey][i] != "-":
                        modes_map[gkey][i] = "-"
                        changed = True
                for a in gargs:
                    bound |= _vars_in(a)
    return changed

    def parse_errors(output: str) -> list[tuple[str, int, str, str]]:
    """Return list of (pred_name, arity, unbound_var, clause_text)."""
    errors = []
    pred_name, arity = None, None
    lines = output.splitlines()
    i = 0
    while i < len(lines):
        line = lines[i]
        m = PRED_HDR_RE.search(line)
        if m:
            pred_name = m.group(1)
            arity = int(m.group(2))
        m = ERR_RE.match(line.rstrip())
        if m and pred_name is not None:
            var = m.group(1)
            clause = lines[i + 1].strip() if i + 1 < len(lines) else ""
            errors.append((pred_name, arity, var, clause))
        i += 1
    return errors

    def demote_once(pl_file: Path) -> tuple[int, int, int]:
    """Run CasLog, parse errors, apply demotions.
    Returns (n_errors_before, n_demotions_applied, n_errors_after).
    """
    text = pl_file.read_text()
    modes = read_modes(text)
    output = run_caslog(pl_file)
    errs = parse_errors(output)
    n_before = len(errs)
    bounded_before = output.count("Time functions for predicate")
    demotions: set[tuple[str, int, int]] = set()

    for pred_name, arity, var, clause in errs:
        head, body, _ = split_head_body(clause)
        hname, hargs = parse_head(head)

        if body:
            for cname, cargs in parse_body_calls(body):
                key = (cname, len(cargs))
                cmodes = modes.get(key)
                if cmodes is None:
                    continue
                for i in var_positions_at_depth0(cargs, var):
                    if i < len(cmodes) and cmodes[i] == "+":
                        demotions.add((cname, len(cargs), i))

        if hname:
            hkey = (hname, len(hargs))
            hmodes = modes.get(hkey)
            if hmodes is not None:
                for i in var_positions_at_depth0(hargs, var):
                    if i < len(hmodes) and hmodes[i] == "+":
                        demotions.add((hname, len(hargs), i))

    if not demotions:
        return n_before, 0, n_before

    modes_batch = {k: list(v) for k, v in modes.items()}
    for (name, arity, i) in demotions:
        m = modes_batch[(name, arity)]
        if i < len(m) and m[i] == "+":
            m[i] = "-"
    pl_file.write_text(write_modes(text, modes_batch))
    out_b = run_caslog(pl_file)
    errs_b = parse_errors(out_b)
    bnd_b = out_b.count("Time functions for predicate")
    all_aux = all(n.startswith(("aux_", "loop_aux_")) for n, _, _ in demotions)
    if all_aux or bnd_b >= bounded_before:
        return n_before, len(demotions), len(errs_b)

    pl_file.write_text(text)
    current_text = text
    current_bnd = bounded_before
    current_errs = n_before
    kept = 0
    for (name, arity, i) in sorted(demotions):
        trial_modes = read_modes(current_text)
        key = (name, arity)
        if key not in trial_modes:
            continue
        m = trial_modes[key]
        if not (i < len(m) and m[i] == "+"):
            continue
        m[i] = "-"
        trial_text = write_modes(current_text, trial_modes)
        pl_file.write_text(trial_text)
        out_t = run_caslog(pl_file)
        bnd_t = out_t.count("Time functions for predicate")
        errs_t = len(parse_errors(out_t))
        is_aux = name.startswith(("aux_", "loop_aux_"))
        if is_aux or bnd_t >= current_bnd:
            current_text = trial_text
            current_bnd = bnd_t
            current_errs = errs_t
            kept += 1
        else:
            pl_file.write_text(current_text)
    return n_before, kept, current_errs
